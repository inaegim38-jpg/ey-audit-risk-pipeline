// Power BI DAX measures — 포렌식 계층
// 모델 매핑
//   Evidence      ← forensic.vw_pbi_evidence
//   Documents     ← forensic.vw_pbi_review_progress
//   Findings      ← forensic.vw_pbi_findings
//   CaseSummary   ← forensic.vw_pbi_case_summary
// 전제: sql/08_forensic_case_schema.sql, sql/09_forensic_pbi_views.sql 실행 후

// ---------------------------------------------------------------- 증거
Evidence Count = SUM ( Evidence[evidence_count] )

Evidence Volume (MB) = DIVIDE ( SUM ( Evidence[total_bytes] ), 1024 * 1024, 0 )

Unique Hash Ratio (%) =
DIVIDE ( SUM ( Evidence[unique_hashes] ), SUM ( Evidence[evidence_count] ), 0 ) * 100

// 수집 시각 범위는 슬라이서용 날짜 테이블로 분리하는 것을 권장
Collection Start = MIN ( Evidence[first_collected_at] )
Collection End = MAX ( Evidence[last_collected_at] )

// ---------------------------------------------------------------- 문서 검토
Documents Processed = SUM ( Documents[doc_count] )

Review Progress (%) =
DIVIDE (
    CALCULATE ( SUM ( Documents[doc_count] ), Documents[review_status] <> "UNREVIEWED" ),
    SUM ( Documents[doc_count] ),
    0
) * 100

High Priority Docs =
CALCULATE ( SUM ( Documents[doc_count] ), Documents[suggested_relevance] = "HIGH" )

Near Dup Share (%) =
DIVIDE ( SUM ( Documents[near_dup_docs] ), SUM ( Documents[doc_count] ), 0 ) * 100

Avg Priority = AVERAGE ( Documents[avg_priority] )

// ---------------------------------------------------------------- 파인딩
Findings Count = COUNTROWS ( Findings )

Critical Findings =
CALCULATE ( COUNTROWS ( Findings ), Findings[severity] = "CRITICAL" )

Open Critical Findings =
CALCULATE (
    COUNTROWS ( Findings ),
    Findings[severity] = "CRITICAL",
    Findings[status] IN { "OPEN", "IN_REVIEW" }
)

Finding Amount = SUM ( Findings[amount] )

// 증거가 하나라도 연결된 파인딩 비율. 인용 없는 파인딩은 검토 대상에서 빠졌다는 뜻
Evidence Citation Coverage (%) =
DIVIDE (
    CALCULATE ( COUNTROWS ( Findings ), Findings[evidence_count] > 0 ),
    COUNTROWS ( Findings ),
    0
) * 100

Avg Evidence per Finding =
DIVIDE ( SUM ( Findings[evidence_count] ), COUNTROWS ( Findings ), 0 )

// ---------------------------------------------------------------- 카드 1장용
Case Evidence Count = SUM ( CaseSummary[evidence_count] )
Case Document Count = SUM ( CaseSummary[document_count] )
Case Finding Count = SUM ( CaseSummary[finding_count] )
Case Critical Count = SUM ( CaseSummary[critical_findings] )
Case Open Count = SUM ( CaseSummary[open_findings] )
Case Total Amount = SUM ( CaseSummary[total_finding_amount] )

// ---------------------------------------------------------------- 참고
// 1) Documents 는 이미 집계된 뷰다. 문서 1건 단위 슬라이서가 필요하면
//    forensic.documents 를 직접 가져오고 측정값은 COUNTROWS 로 바꾼다.
// 2) review_status 값은 UNREVIEWED / REVIEWED / ESCALATED 로 운영하는 것을 권장.
//    현재 파이프라인은 UNREVIEWED 만 생성하고, 사람이 검토 콘솔에서 갱신한다.
// 3) DIVIDE 의 세 번째 인자(0)를 생략하면 BLANK 가 되어 카드가 비어 보인다.
